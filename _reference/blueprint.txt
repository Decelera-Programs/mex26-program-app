function getEventTypeColor(type) {
  const colors = {
    workshop: "from-blue-500 to-indigo-600",
    talk: "from-violet-500 to-purple-600",
    networking: "from-amber-500 to-orange-600",
    meal: "from-emerald-500 to-green-600",
    activity: "from-cyan-500 to-teal-600",
    ceremony: "from-rose-500 to-pink-600",
    mentoring: "from-sky-500 to-blue-600",
    free_time: "from-gray-400 to-gray-500"
  };
  return colors[type] || "from-primary to-primary";
}


export default function NextEventCard({ event }) {
  const [countdown, setCountdown] = useState("");


  useEffect(() => {
    if (!event) return;
    const interval = setInterval(() => {
      const now = moment();
      const start = moment(event.start_time);
      const diff = start.diff(now);


      if (diff <= 0) {
        const end = moment(event.end_time);
        if (now.isBefore(end)) {
          setCountdown("Happening now");
        } else {
          setCountdown("Ended");
        }
      } else {
        const duration = moment.duration(diff);
        if (duration.asHours() >= 1) {
          setCountdown(`Starts in ${Math.floor(duration.asHours())}h ${duration.minutes()}m`);
        } else {
          setCountdown(`Starts in ${duration.minutes()}m`);
        }
      }
    }, 1000);
    return () => clearInterval(interval);
  }, [event]);


  if (!event) {
    return (
      <div className="bg-card rounded-2xl p-6 border border-border/50">
        <p className="text-muted-foreground text-sm">No upcoming events today</p>
      </div>);


  }


  const isHappening = countdown === "Happening now";


  return (
    <Link to={`/event/${event.id}`}>
      <motion.div
        initial={{ opacity: 0, y: 20 }}
        animate={{ opacity: 1, y: 0 }}
        transition={{ duration: 0.5, ease: "easeOut" }} className="bg-[hsl(var(--sidebar-ring))] text-white p-6 rounded-3xl relative overflow-hidden from-amber-500 to-orange-600 shadow-lg">


       
        {/* Decorative circles */}
        <div className="absolute -top-8 -right-8 w-32 h-32 rounded-full bg-white/10" />
        <div className="absolute -bottom-4 -left-4 w-20 h-20 rounded-full bg-white/5" />


        <div className="relative z-10">
          <div className="flex items-center justify-between mb-4">
            <span className={`inline-flex items-center gap-1.5 px-3 py-1 rounded-full text-xs font-semibold ${
            isHappening ? "bg-white/30 animate-pulse" : "bg-white/20"}`
            }>
              <span className={`w-1.5 h-1.5 rounded-full ${isHappening ? "bg-green-300" : "bg-white/80"}`} />
              {countdown}
            </span>
            <ArrowRight className="h-4 w-4 opacity-60" />
          </div>


          <h3 className="text-xl font-bold mb-3 leading-tight">{event.title}</h3>


          <div className="flex items-center gap-4 text-white/80 text-sm">
            <span className="flex items-center gap-1.5">

              <MapPin className="h-3.5 w-3.5" />

            </span>

import { useState, useEffect } from "react";
import { base44 } from "@/api/base44Client";


export function useUserSchedule() {
  const [events, setEvents] = useState([]);
  const [user, setUser] = useState(null);
  const [loading, setLoading] = useState(true);


  useEffect(() => {
    async function fetchData() {
      setLoading(true);
      const me = await base44.auth.me();
      setUser(me);


      // Get all user-event assignments for this user
      const userEvents = await base44.entities.UserEvent.filter(
        { user_email: me.email },
        "created_date"
      );


      if (userEvents.length > 0) {
        // Fetch all events
        const allEvents = await base44.entities.Event.list("start_time", 200);
        const assignedIds = new Set(userEvents.map((ue) => ue.event_id));
        const myEvents = allEvents.filter((e) => assignedIds.has(e.id));
        setEvents(myEvents);
      }


      setLoading(false);
    }
    fetchData();
  }, []);


  return { events, user, loading };
}

  const [viewMode, setViewMode] = useState("detailed"); // "detailed" | "compact"

  const bootcampStart = useMemo(() => {
    if (events.length === 0) return moment().startOf("day");
    const sorted = [...events].sort((a, b) => moment(a.start_time).diff(moment(b.start_time)));
    return moment(sorted[0].start_time).startOf("day");
  }, [events]);

    return Array.from({ length: 7 }, (_, i) =>
      moment(bootcampStart).add(i, "days")
    );
  }, [bootcampStart]);

    return events
      .filter((e) => moment(e.start_time).isSame(selectedDate, "day"))
      .sort((a, b) => moment(a.start_time).diff(moment(b.start_time)));

import { useNavigate, useParams, Link } from "react-router-dom";
import { ArrowLeft, Globe, MapPin, Calendar } from "lucide-react";
import { motion } from "framer-motion";
import { base44 } from "@/api/base44Client";


const stageColors = {
  "pre-seed": "bg-slate-100 text-slate-600",
  "seed": "bg-blue-100 text-blue-700",
  "series-a": "bg-violet-100 text-violet-700",
  "series-b": "bg-amber-100 text-amber-700",
};


export default function StartupDetail() {
  const { id } = useParams();
  const navigate = useNavigate();
  const [startup, setStartup] = useState(null);
  const [founders, setFounders] = useState([]);
  const [loading, setLoading] = useState(true);


  useEffect(() => {
    async function fetch() {
      const [startups, people] = await Promise.all([
        base44.entities.Startup.list("name", 50),
        base44.entities.Person.list("full_name", 200),
      ]);
      const found = startups.find((s) => s.id === id);
      setStartup(found || null);
      if (found) {
        setFounders(people.filter((p) => p.startup_id === found.id && p.person_type === "founder"));
      }
      setLoading(false);
    }
    fetch();
  }, [id]);


  if (loading) {
    return (
      <div className="flex items-center justify-center h-screen">
        <div className="w-8 h-8 border-4 border-primary/20 border-t-primary rounded-full animate-spin" />
      </div>
    );
  }


  if (!startup) {
    return (
      <div className="flex flex-col items-center justify-center h-screen px-5">
        <p className="text-4xl mb-4">🚀</p>
        <p className="text-muted-foreground">Startup not found</p>
        <button onClick={() => navigate(-1)} className="mt-4 text-primary text-sm font-medium">Go Back</button>
      </div>
    );
  }


  return (
    <div className="min-h-screen">
      {/* Hero */}
      <div className="relative bg-gradient-to-br from-primary via-blue-600 to-indigo-700 px-5 pt-12 pb-20">
        <div className="absolute top-0 right-0 w-48 h-48 rounded-full bg-white/5" />
        <motion.button
          initial={{ opacity: 0 }}
          animate={{ opacity: 1 }}
          onClick={() => navigate(-1)}
          className="flex items-center gap-2 text-white/80 hover:text-white mb-8"
        >
          <ArrowLeft className="h-5 w-5" />
          <span className="text-sm font-medium">Back</span>
        </motion.button>


        <motion.div
          initial={{ opacity: 0, y: 20 }}
          animate={{ opacity: 1, y: 0 }}
          transition={{ delay: 0.1 }}
          className="relative z-10 flex flex-col items-center text-center"
        >
          {startup.logo_url ? (
            <img src={startup.logo_url} alt={startup.name}
              className="w-20 h-20 rounded-2xl object-cover border-4 border-white/30 shadow-xl mb-4" />
          ) : (
            <div className="w-20 h-20 rounded-2xl bg-white/20 border-4 border-white/30 flex items-center justify-center text-white text-3xl font-bold mb-4">
              {startup.name[0]}
            </div>
          )}
          <h1 className="text-2xl font-bold text-white">{startup.name}</h1>
          {startup.tagline && <p className="text-white/70 text-sm mt-2 max-w-xs">{startup.tagline}</p>}
          <div className="flex gap-2 mt-3">
            {startup.stage && (
              <span className="px-3 py-1.5 rounded-full bg-white/20 text-white text-xs font-semibold capitalize">{startup.stage}</span>
            )}
            {startup.sector && (
              <span className="px-3 py-1.5 rounded-full bg-white/10 text-white/80 text-xs">{startup.sector}</span>
            )}
          </div>
        </motion.div>
      </div>


      {/* Card */}
      <motion.div
        initial={{ opacity: 0, y: 20 }}
        animate={{ opacity: 1, y: 0 }}
        transition={{ delay: 0.2 }}
        className="relative -mt-10 mx-4 bg-card rounded-3xl border border-border/50 shadow-xl"
      >
        <div className="p-6 space-y-5">
          {/* Meta */}
          <div className="flex gap-4">
            {startup.hq_location && (
              <div className="flex items-center gap-1.5 text-xs text-muted-foreground">
                <MapPin className="h-3.5 w-3.5" />
                {startup.hq_location}
              </div>
            )}
            {startup.founding_year && (
              <div className="flex items-center gap-1.5 text-xs text-muted-foreground">
                <Calendar className="h-3.5 w-3.5" />
                Founded {startup.founding_year}
              </div>
            )}
          </div>


          {startup.website_url && (
            <a href={startup.website_url} target="_blank" rel="noopener noreferrer"
              className="flex items-center gap-2 text-sm font-medium text-primary bg-primary/10 px-4 py-2.5 rounded-xl w-fit">
              <Globe className="h-4 w-4" />
              Visit Website
            </a>
          )}


          {startup.description && (
            <>
              <div className="border-t border-border/50" />
              <div>
                <h3 className="text-xs font-semibold text-muted-foreground uppercase tracking-widest mb-2">About</h3>
                <p className="text-sm text-foreground/80 leading-relaxed">{startup.description}</p>
              </div>
            </>
          )}


          {/* Founders */}
          {founders.length > 0 && (
            <>
              <div className="border-t border-border/50" />
              <div>
                <h3 className="text-xs font-semibold text-muted-foreground uppercase tracking-widest mb-3">Founders</h3>
                <div className="space-y-2">
                  {founders.map((founder) => {
                    const initials = founder.full_name.split(" ").map((n) => n[0]).join("").toUpperCase().slice(0, 2);
                    return (
                      <Link key={founder.id} to={`/person/${founder.id}`}
                        className="flex items-center gap-3 bg-secondary rounded-2xl p-3 hover:bg-secondary/80 transition-colors">
                        {founder.photo_url ? (
                          <img src={founder.photo_url} alt={founder.full_name} className="w-10 h-10 rounded-xl object-cover" />
                        ) : (
                          <div className="w-10 h-10 rounded-xl bg-rose-100 flex items-center justify-center text-rose-600 font-bold text-sm">
                            {initials}
                          </div>
                        )}
                        <div>
                          <p className="text-sm font-semibold">{founder.full_name}</p>
                          {founder.title && <p className="text-xs text-muted-foreground">{founder.title}</p>}
                        </div>
                      </Link>
                    );
                  })}
                </div>
              </div>
            </>
          )}
        </div>
      </motion.div>
      <div className="h-8" />
    </div>
  );
}

Startups.jsx:
import { useState, useEffect } from "react";
import { Search, ExternalLink } from "lucide-react";
import { Link } from "react-router-dom";
import { motion } from "framer-motion";
import { base44 } from "@/api/base44Client";


const stageColors = {
  "pre-seed": "bg-slate-100 text-slate-600",
  "seed": "bg-blue-100 text-blue-700",
  "series-a": "bg-violet-100 text-violet-700",
  "series-b": "bg-amber-100 text-amber-700",
};


export default function Startups() {
  const [startups, setStartups] = useState([]);
  const [people, setPeople] = useState([]);
  const [loading, setLoading] = useState(true);
  const [search, setSearch] = useState("");


  useEffect(() => {
    async function fetch() {
      const [s, p] = await Promise.all([
        base44.entities.Startup.list("name", 50),
        base44.entities.Person.list("full_name", 200),
      ]);
      setStartups(s);
      setPeople(p);
      setLoading(false);
    }
    fetch();
  }, []);


  const filtered = startups.filter(
    (s) =>
      !search ||
      s.name?.toLowerCase().includes(search.toLowerCase()) ||
      s.sector?.toLowerCase().includes(search.toLowerCase()) ||
      s.tagline?.toLowerCase().includes(search.toLowerCase())
  );


  function getFounders(startupId) {
    return people.filter((p) => p.startup_id === startupId && p.person_type === "founder");
  }


  return (
    <div className="px-5 pt-14 pb-6">
      <motion.div initial={{ opacity: 0, y: -10 }} animate={{ opacity: 1, y: 0 }} className="mb-6">
        <h1 className="text-2xl font-bold text-foreground">Startups</h1>
        <p className="text-muted-foreground text-xs mt-0.5">{startups.length} companies at Menorca 2026</p>
      </motion.div>


      <div className="relative mb-6">
        <Search className="absolute left-3.5 top-1/2 -translate-y-1/2 h-4 w-4 text-muted-foreground" />
        <input
          type="text"
          placeholder="Search startups..."
          value={search}
          onChange={(e) => setSearch(e.target.value)}
          className="w-full bg-card border border-border/50 rounded-2xl pl-10 pr-4 py-3 text-sm outline-none focus:border-primary/50 transition-colors"
        />
      </div>


      {loading ? (
        <div className="flex justify-center py-16">
          <div className="w-7 h-7 border-4 border-primary/20 border-t-primary rounded-full animate-spin" />
        </div>
      ) : (
        <div className="space-y-3">
          {filtered.map((startup, i) => {
            const founders = getFounders(startup.id);
            return (
              <motion.div
                key={startup.id}
                initial={{ opacity: 0, y: 12 }}
                animate={{ opacity: 1, y: 0 }}
                transition={{ delay: i * 0.05 }}
              >
                <Link to={`/startup/${startup.id}`} className="block">
                  <div className="bg-card rounded-2xl border border-border/50 p-5 hover:shadow-md transition-all active:scale-[0.98]">
                    <div className="flex items-start gap-4">
                      {startup.logo_url ? (
                        <img src={startup.logo_url} alt={startup.name} className="w-14 h-14 rounded-2xl object-cover flex-shrink-0" />
                      ) : (
                        <div className="w-14 h-14 rounded-2xl bg-gradient-to-br from-primary/20 to-primary/40 flex items-center justify-center text-primary font-bold text-xl flex-shrink-0">
                          {startup.name[0]}
                        </div>
                      )}
                      <div className="flex-1 min-w-0">
                        <div className="flex items-start justify-between gap-2">
                          <p className="font-semibold text-foreground">{startup.name}</p>
                          {startup.stage && (
                            <span className={`text-[10px] font-semibold px-2 py-1 rounded-full flex-shrink-0 ${stageColors[startup.stage] || "bg-muted text-muted-foreground"}`}>
                              {startup.stage}
                            </span>
                          )}
                        </div>
                        {startup.tagline && (
                          <p className="text-xs text-muted-foreground mt-1">{startup.tagline}</p>
                        )}
                        {startup.sector && (
                          <p className="text-xs text-primary/70 mt-1.5 font-medium">{startup.sector}</p>
                        )}
                        {founders.length > 0 && (
                          <div className="flex items-center gap-1.5 mt-2.5">
                            {founders.map((f) => (
                              <span key={f.id} className="text-xs bg-secondary text-secondary-foreground px-2.5 py-1 rounded-full">
                                {f.full_name.split(" ")[0]}
                              </span>
                            ))}
                          </div>
                        )}
                      </div>
                    </div>
                  </div>
                </Link>
              </motion.div>
            );
          })}
        </div>
      )}
    </div>
  );
}
